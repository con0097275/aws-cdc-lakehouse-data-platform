from cdc.metadata_plane import load

m = UrnMinter(load().environment, default_catalog()); g = build_static_graph(m)
s = LineageImpactService(g, environment='dev', minter=m)
inc = Incident('inc-demo', 'eod:oracle_coredb_corebank_account', FailureClass.DQ_VIOLATION,
               'BLOCKER', scope=AffectedScope(cob_dates=(date(2026,9,28),)),
               root_cause_status=RootCauseStatus.CONFIRMED)

                   policy=RecoveryPolicy(max_attempts=2, max_blast_radius=40),
                   lineage_version='lin-demo')

print('blast radius   :', p.blast_radius)
print('turns          :', [len(t) for t in p.turns])
print('weakest edge   :', p.weakest_evidence.value)
print('cost class     :', p.cost_class.value)
print('approval       :', a.value)
for r in why: print('   reason:', r)
print()
