=== oracle service in compose
  oracle:
    image: ${ORACLE_IMAGE}
    container_name: source-oracle
    restart: unless-stopped
    # Bind to the PRIVATE IP only. Publishing on 0.0.0.0 would expose 1521 on the public
    # interface regardless of the security group — defence in depth (CLAUDE.md 3.3).
    ports:
      - "${PRIVATE_IP}:1521:1521"
    environment:
      ORACLE_PWD: ${ORACLE_PASSWORD}
      ORACLE_CHARACTERSET: AL32UTF8
      # Oracle Free is capped at 2 GB by licence; asking for more is silently ignored.
      INIT_SGA_SIZE: "1536"
